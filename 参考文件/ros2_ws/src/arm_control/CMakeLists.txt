cmake_minimum_required(VERSION 3.16)
project(arm_control LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
option(ARM_STANDALONE "Build portable CAN and safety tests without ROS" OFF)
option(ARM_WITH_IGH "Build real IgH EtherCAT transport" OFF)
option(ARM_WITH_ZCAN "Build against vendor ZCAN header/library" OFF)
option(ARM_WITH_LOONG "Build against the patched Loong driver SDK" OFF)
add_library(arm_bus STATIC src/bus.cpp src/can_bus.cpp src/mit_bus.cpp src/can_driver.cpp src/robomaster_bus.cpp)
find_package(Threads REQUIRED)
target_link_libraries(arm_bus PUBLIC Threads::Threads)
set_target_properties(arm_bus PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(arm_bus PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(arm_bus PRIVATE src/socketcan_driver.cpp)
endif()
if(ARM_WITH_IGH)
  find_path(IGH_INCLUDE_DIR ecrt.h REQUIRED)
  find_library(IGH_LIBRARY ethercat REQUIRED)
  target_sources(arm_bus PRIVATE src/igh_bus.cpp)
  target_include_directories(arm_bus PRIVATE ${IGH_INCLUDE_DIR})
  target_link_libraries(arm_bus PUBLIC ${IGH_LIBRARY})
  target_compile_definitions(arm_bus PRIVATE ARM_WITH_IGH)
endif()
if(ARM_WITH_ZCAN)
  find_path(ZCAN_INCLUDE_DIR zlgcan.h REQUIRED)
  find_library(ZCAN_LIBRARY NAMES zlgcan REQUIRED)
  target_sources(arm_bus PRIVATE src/zcan_driver.cpp)
  target_include_directories(arm_bus PRIVATE ${ZCAN_INCLUDE_DIR})
  target_link_libraries(arm_bus PUBLIC ${ZCAN_LIBRARY})
  target_compile_definitions(arm_bus PRIVATE ARM_WITH_ZCAN)
endif()
if(ARM_WITH_LOONG)
  find_path(LOONG_INCLUDE_DIR loong_driver_sdk.h REQUIRED)
  find_library(LOONG_LIBRARY NAMES loong_driver_sdk_x86_64 loong_driver_sdk_aarch64 loong_driver_sdk REQUIRED)
  find_package(tinyxml2 REQUIRED)
  find_package(Threads REQUIRED)
  target_sources(arm_bus PRIVATE src/loong_bus.cpp)
  target_include_directories(arm_bus PRIVATE ${LOONG_INCLUDE_DIR})
  target_link_libraries(arm_bus PUBLIC ${LOONG_LIBRARY} tinyxml2::tinyxml2 Threads::Threads)
  target_compile_definitions(arm_bus PRIVATE ARM_WITH_LOONG)
endif()
if(BUILD_TESTING)
  foreach(test safety can robomaster mit)
    add_executable(test_${test} test/test_${test}.cpp)
    target_link_libraries(test_${test} arm_bus)
    add_test(NAME ${test}_core COMMAND test_${test})
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME configuration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_config.py)
  add_test(NAME kinematics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_kinematics.py)
  add_test(NAME trajectory_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_trajectory.py)
  add_test(NAME recovery_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_recover.py)
  add_test(NAME continuous_path COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_path.py)
  add_executable(trajectory_wire test/trajectory_wire.cpp)
  target_link_libraries(trajectory_wire arm_bus)
  add_test(NAME wire_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_wire_pipeline.py $<TARGET_FILE:trajectory_wire>)
endif()
if(ARM_STANDALONE)
  return()
endif()
find_package(ament_cmake REQUIRED)
foreach(dep hardware_interface pluginlib rclcpp rclcpp_lifecycle diagnostic_msgs std_srvs std_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(arm_system SHARED src/system.cpp)
target_link_libraries(arm_system arm_bus)
ament_target_dependencies(arm_system hardware_interface pluginlib rclcpp rclcpp_lifecycle diagnostic_msgs std_srvs std_msgs)
pluginlib_export_plugin_description_file(hardware_interface arm_plugins.xml)
install(TARGETS arm_system arm_bus EXPORT arm_system DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(PROGRAMS scripts/latency_probe.py scripts/send_trajectory.py scripts/recover.py scripts/move_pose.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/kinematics.py scripts/trajectory_path.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_system test/test_system.cpp)
  target_link_libraries(test_system arm_system)
  ament_target_dependencies(test_system hardware_interface rclcpp rclcpp_lifecycle diagnostic_msgs std_srvs std_msgs)
  add_test(NAME system_lifecycle COMMAND test_system)
endif()
ament_export_targets(arm_system HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle diagnostic_msgs std_srvs std_msgs)
ament_package()
